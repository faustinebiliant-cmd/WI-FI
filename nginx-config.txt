server {
    listen 80;
    listen [::]:80;
    server_name oswagolink.co.tz www.oswagolink.co.tz;

    # Redirect all HTTP traffic to HTTPS
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    listen [::]:443 ssl;
    server_name oswagolink.co.tz www.oswagolink.co.tz;

    # SSL Certificates managed by Certbot
    ssl_certificate /etc/letsencrypt/live/oswagolink.co.tz/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/oswagolink.co.tz/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # API Proxy (Backend)
    location /api/ {
        proxy_pass http://127.0.0.1:5001/api/;
        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Root Domain Redirect (Goes to Tenant Login for now)
    location = / {
        return 301 https://oswagolink.co.tz/tenant/login;
    }

    # Portal (Guest App)
    location = /portal {
        return 301 /portal/;
    }
    location /portal/ {
        alias /var/www/wifi-app/portal/;
        try_files $uri $uri/ /portal/index.html;
    }

    # Tenant Panel (Hotel App)
    location = /tenant {
        return 301 /tenant/;
    }
    location /tenant/ {
        alias /var/www/wifi-app/tenant/;
        try_files $uri $uri/ /tenant/index.html;
    }

    # Platform Panel (Admin App)
    location = /platform {
        return 301 /platform/;
    }
    location /platform/ {
        alias /var/www/wifi-app/platform/;
        try_files $uri $uri/ /platform/index.html;
    }
}
